function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("AppState") || ss.insertSheet("AppState");
  var raw = sheet.getRange("A1").getValue();
  return ContentService.createTextOutput(raw || "{}")
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);

  // 1. Save full live state (Settings, Courses, Photos, Videos, Reviews)
  var stateSheet = ss.getSheetByName("AppState") || ss.insertSheet("AppState");
  if (payload.state) {
    stateSheet.getRange("A1").setValue(JSON.stringify(payload.state));
  }

  // 2. Append every new student booking into a clean readable "Bookings" tab
  if (payload.booking) {
    var bkSheet = ss.getSheetByName("Bookings") || ss.insertSheet("Bookings");
    if (bkSheet.getLastRow() === 0) {
      bkSheet.appendRow(["Booking ID", "Date", "Student Name", "Mobile", "Course", "Car", "Batch Time", "Bhiwadi Area"]);
    }
    var b = payload.booking;
    bkSheet.appendRow([b.id, b.date, b.name, b.phone, b.course, b.car, b.time, b.area]);
  }

  return ContentService.createTextOutput(JSON.stringify({ status: "OK" }))
    .setMimeType(ContentService.MimeType.JSON);
}
